import com.biletflow.biletflow.ticketinventory.application.eventmanagement.port.TicketEventDetails;

import com.biletflow.biletflow.ticketinventory.domain.ticket.TicketStatus;

/** Current presentation of a canonical ticket; rendering never issues a new ticket or code. */
public record TicketDocument(TicketView ticket, String ticketType, TicketEventDetails event) {
    public boolean hasAdmissionCode() {
        return !event.cancelled() && (ticket.status() == TicketStatus.VALID || ticket.status() == TicketStatus.CHECKED_IN);
    }

    public String displayManualCode() {
        String code = ticket.manualCode();
        if (!hasAdmissionCode() || code == null) return null;
        return code.substring(0, 4) + "-" + code.substring(4);
    }
}

@Service
@Transactional(readOnly = true)
public class TicketDocumentService {

    private final TicketRepository tickets;
    private final TicketTypeRepository types;
    private final EventManagementPort events;
    private final VerifiedUserEmailPort emails;
    private final CurrentActor actor;
    private final TicketDocumentRenderer renderer;
    private final TicketDeliveryStore deliveries;
    private final Clock clock;

    public TicketDocumentService(
        TicketRepository tickets,
        TicketTypeRepository types,
        EventManagementPort events,
        VerifiedUserEmailPort emails,
        CurrentActor actor,
        TicketDocumentRenderer renderer,
        TicketDeliveryStore deliveries,
        Clock clock
    ) {
        this.tickets = Objects.requireNonNull(tickets);
        this.types = Objects.requireNonNull(types);
        this.events = Objects.requireNonNull(events);
        this.emails = Objects.requireNonNull(emails);
        this.actor = Objects.requireNonNull(actor);
        this.renderer = Objects.requireNonNull(renderer);
        this.deliveries = Objects.requireNonNull(deliveries);
        this.clock = Objects.requireNonNull(clock);
    }

    public byte[] pdf(UUID ticketId) {
        return renderer.pdf(document(authorize(ticketId)));
    }

    public byte[] qr(UUID ticketId) {
        return renderer.qr(document(authorize(ticketId)));
    }

    @Transactional
    public void requestEmail(UUID ticketId) {
        var ticket = authorize(ticketId);
        var document = document(ticket);
        if (ticket.getStatus() != TicketStatus.VALID || document.event().cancelled()) {
            throw new IllegalStateException("Only a valid ticket for a non-cancelled event can be emailed");
        }
        deliveries.request(ticketId, clock.instant());
    }

    /** Internal worker query; no public endpoint delegates directly to this method. */
    public TicketDocument forDelivery(UUID ticketId) {
        return document(requireTicket(ticketId));
    }

    private Ticket authorize(UUID ticketId) {
        Long user = actor.requireUserId();
        var ticket = requireTicket(ticketId);
        boolean recipient = emails
            .findVerifiedEmail(user)
            .map(email -> new AttendeeEmail(email).equals(ticket.getAttendeeEmail()))
            .orElse(false);
        if (!user.equals(ticket.getOwnerUserId()) && !recipient && !events.canManageTicketing(ticket.getEventId(), user)) {
            throw new AuthorizationException("Ticket is private");
        }
        return ticket;
    }

    private Ticket requireTicket(UUID id) {
        return tickets.findById(new TicketId(id)).orElseThrow(() -> new EntityNotFoundException("Ticket not found: " + id));
    }

    private TicketDocument document(Ticket ticket) {
        var type = types.findById(ticket.getTicketTypeId()).orElseThrow(() -> new EntityNotFoundException("Ticket type not found"));
        var view = new TicketView(
            ticket.getId().value(),
            ticket.getTicketTypeId().value(),
            ticket.getEventId(),
            ticket.getOrderId().value(),
            ticket.getAttendeeEmail().value(),
            ticket.getOwnerUserId(),
            ticket.getSeatId() == null ? null : ticket.getSeatId().value(),
            ticket.getTicketCode().value(),
            ticket.getManualCode(),
            ticket.getStatus(),
            ticket.getIssuedAt()
        );
        return new TicketDocument(view, type.getName(), events.ticketDetails(ticket.getEventId(), view.seatId()));
    }
}

@Service
public class TicketSaleService {

    private final EventInventoryRepository inventoryRepository;
    private final TicketRepository ticketRepository;
    private final TicketTypeRepository ticketTypeRepository;
    private final Clock clock;
    private final TicketDeliveryStore deliveries;

    public TicketSaleService(
        EventInventoryRepository inventoryRepository,
        TicketRepository ticketRepository,
        TicketTypeRepository ticketTypeRepository,
        Clock clock,
        TicketDeliveryStore deliveries
    ) {
        this.inventoryRepository = Objects.requireNonNull(inventoryRepository);
        this.ticketRepository = Objects.requireNonNull(ticketRepository);
        this.ticketTypeRepository = Objects.requireNonNull(ticketTypeRepository);
        this.clock = Objects.requireNonNull(clock);
        this.deliveries = Objects.requireNonNull(deliveries);
    }

    @Transactional
    public List<TicketView> complete(CompleteSaleCommand command) {
        Objects.requireNonNull(command, "command cannot be null");

        OrderId orderId = new OrderId(command.orderId());

        List<Ticket> existing = ticketRepository.findAllByOrderId(orderId);

        if (!existing.isEmpty()) {
            ensureRetryMatchesExisting(command, existing);

            return existing.stream().map(this::toView).toList();
        }

        EventInventory inventory = inventoryRepository
            .findByEventId(command.eventId())
            .orElseThrow(() -> new EntityNotFoundException("EventInventory not found for event: " + command.eventId()));

        validateHoldItemsMatch(inventory, command.holds(), command.items());

        validateTicketTypes(command.eventId(), command.items());

        Instant now = clock.instant();

        for (HoldReference reference : command.holds()) {
            if (reference instanceof GaReservationReference ga) {
                inventory.confirmGAReservation(new InventoryReservationId(ga.reservationId()), orderId, now);
            } else if (reference instanceof SeatHoldReference seat) {
                inventory.confirmSeatHold(new SeatHoldId(seat.holdId()), orderId, now);
            } else {
                throw new IllegalStateException("Unsupported hold reference: " + reference.getClass().getName());
            }
        }

        List<Ticket> tickets = command
            .items()
            .stream()
            .map(item ->
                Ticket.issue(
                    new TicketTypeId(item.ticketTypeId()),
                    command.eventId(),
                    orderId,
                    new AttendeeEmail(item.attendeeEmail()),
                    item.ownerUserId(),
                    item.seatId() == null ? null : new SeatId(item.seatId()),
                    now
                )
            )
            .toList();

        inventoryRepository.save(inventory);

        var saved = ticketRepository.saveAll(tickets);
        deliveries.enqueue(
            saved
                .stream()
                .map(ticket -> ticket.getId().value())
                .toList(),
            now
        );
        return saved.stream().map(this::toView).toList();
    }

    private void validateHoldItemsMatch(EventInventory inventory, List<HoldReference> holds, List<IssueTicketItem> items) {
        if (inventory.getMode() instanceof GeneralAdmissionInventory ga) {
            Map<TicketTypeId, Integer> heldQuantities = new HashMap<>();

            for (HoldReference reference : holds) {
                if (!(reference instanceof GaReservationReference gaRef)) {
                    throw new IllegalStateException("Assigned-seat hold supplied for general-admission inventory");
                }

                InventoryReservation reservation = ga
                    .getReservations()
                    .stream()
                    .filter(candidate -> candidate.getId().value().equals(gaRef.reservationId()))
                    .findFirst()
                    .orElseThrow(() -> new IllegalStateException("GA reservation does not belong to this inventory"));

                heldQuantities.merge(reservation.getTicketTypeId(), reservation.getQuantity(), Integer::sum);
            }

            Map<TicketTypeId, Integer> itemQuantities = new HashMap<>();

            for (IssueTicketItem item : items) {
                if (item.seatId() != null) {
                    throw new IllegalStateException("General-admission ticket item cannot contain a seat");
                }

                itemQuantities.merge(new TicketTypeId(item.ticketTypeId()), 1, Integer::sum);
            }

            if (!heldQuantities.equals(itemQuantities)) {
                throw new IllegalStateException("Ticket items do not match reserved GA inventory");
            }

            return;
        }

        AssignedSeatingInventory assigned = (AssignedSeatingInventory) inventory.getMode();

        Set<SeatFingerprint> heldSeats = holds
            .stream()
            .map(reference -> {
                if (!(reference instanceof SeatHoldReference seatRef)) {
                    throw new IllegalStateException("GA reservation supplied for assigned-seating inventory");
                }

                SeatHold hold = assigned
                    .getHolds()
                    .stream()
                    .filter(candidate -> candidate.getId().value().equals(seatRef.holdId()))
                    .findFirst()
                    .orElseThrow(() -> new IllegalStateException("Seat hold does not belong to this inventory"));

                return new SeatFingerprint(hold.getTicketTypeId().value(), hold.getSeatId().value());
            })
            .collect(Collectors.toUnmodifiableSet());

        Set<SeatFingerprint> requestedSeats = items
            .stream()
            .map(item -> {
                if (item.seatId() == null) {
                    throw new IllegalStateException("Assigned-seating ticket item requires a seat");
                }

                return new SeatFingerprint(item.ticketTypeId(), item.seatId());
            })
            .collect(Collectors.toUnmodifiableSet());

        if (heldSeats.size() != items.size() || !heldSeats.equals(requestedSeats)) {
            throw new IllegalStateException("Ticket items do not match held assigned seats");
        }
    }

    private void validateTicketTypes(UUID eventId, List<IssueTicketItem> items) {
        for (IssueTicketItem item : items) {
            TicketType ticketType = ticketTypeRepository
                .findById(new TicketTypeId(item.ticketTypeId()))
                .orElseThrow(() -> new EntityNotFoundException("TicketType not found: " + item.ticketTypeId()));

            if (!ticketType.getEventId().equals(eventId)) {
                throw new IllegalStateException("Ticket type does not belong to sale event");
            }
        }
    }

    private void ensureRetryMatchesExisting(CompleteSaleCommand command, List<Ticket> existing) {
        if (existing.size() != command.items().size()) {
            throw new IllegalStateException("Order already has a different number of issued tickets");
        }

        Map<IssueFingerprint, Long> requested = command
            .items()
            .stream()
            .map(item ->
                new IssueFingerprint(
                    item.ticketTypeId(),
                    new AttendeeEmail(item.attendeeEmail()).value(),
                    item.ownerUserId(),
                    item.seatId()
                )
            )
            .collect(Collectors.groupingBy(fingerprint -> fingerprint, Collectors.counting()));

        Map<IssueFingerprint, Long> stored = existing
            .stream()
            .map(ticket ->
                new IssueFingerprint(
                    ticket.getTicketTypeId().value(),
                    ticket.getAttendeeEmail().value(),
                    ticket.getOwnerUserId(),
                    ticket.getSeatId() == null ? null : ticket.getSeatId().value()
                )
            )
            .collect(Collectors.groupingBy(fingerprint -> fingerprint, Collectors.counting()));

        if (!requested.equals(stored)) {
            throw new IllegalStateException("Order already has tickets that do not match this sale retry");
        }
    }

    private TicketView toView(Ticket ticket) {
        return new TicketView(
            ticket.getId().value(),
            ticket.getTicketTypeId().value(),
            ticket.getEventId(),
            ticket.getOrderId().value(),
            ticket.getAttendeeEmail().value(),
            ticket.getOwnerUserId(),
            ticket.getSeatId() == null ? null : ticket.getSeatId().value(),
            ticket.getTicketCode().value(),
            ticket.getManualCode(),
            ticket.getStatus(),
            ticket.getIssuedAt()
        );
    }

    private record SeatFingerprint(UUID ticketTypeId, UUID seatId) {}

    private record IssueFingerprint(UUID ticketTypeId, String attendeeEmail, Long ownerUserId, UUID seatId) {}
}

import com.biletflow.biletflow.common.application.AuthorizationException;

import java.util.List;

import java.util.Objects;
import java.util.UUID;
import java.util.regex.Pattern;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@Transactional(readOnly = true)
public class CheckInService {

    private static final Pattern QR_CODE = Pattern.compile("[0-9a-fA-F]{8}(?:-[0-9a-fA-F]{4}){3}-[0-9a-fA-F]{12}");
    private static final Pattern MANUAL_CODE = Pattern.compile("[0-9A-HJKMNP-TV-Z]{8}");

    private final EventManagementPort events;
    private final CurrentActor actor;
    private final TicketService tickets;
    private final TicketRepository repository;
    private final TicketInventoryAuditService audit;

    public CheckInService(
        EventManagementPort events,
        CurrentActor actor,
        TicketService tickets,
        TicketRepository repository,
        TicketInventoryAuditService audit
    ) {
        this.events = Objects.requireNonNull(events);
        this.actor = Objects.requireNonNull(actor);
        this.tickets = Objects.requireNonNull(tickets);
        this.repository = Objects.requireNonNull(repository);
        this.audit = Objects.requireNonNull(audit);
    }

    public TicketValidationResult validate(UUID eventId, String ticketCode) {
        authorize(eventId);
        return tickets.validate(resolveTicketCode(eventId, ticketCode), eventId);
    }

    @Transactional
    public TicketView checkIn(UUID eventId, String ticketCode) {
        Long staffUserId = authorize(eventId);
        TicketView ticket = tickets.checkIn(new CheckInTicketCommand(resolveTicketCode(eventId, ticketCode), eventId));
        audit.record(eventId, staffUserId, TicketInventoryAuditType.TICKET_CHECKED_IN, ticket.id(), "Ticket checked in");
        return ticket;
    }

    @Transactional
    public TicketView reverse(UUID eventId, String ticketCode) {
        Long staffUserId = authorize(eventId);
        TicketView ticket = tickets.reverseCheckIn(new ReverseCheckInCommand(resolveTicketCode(eventId, ticketCode), eventId));
        audit.record(eventId, staffUserId, TicketInventoryAuditType.CHECK_IN_REVERSED, ticket.id(), "Check-in reversed");
        return ticket;
    }

    public List<TicketView> search(UUID eventId, String attendeeEmail) {
        authorize(eventId);
        if (attendeeEmail == null || attendeeEmail.trim().length() < 3) {
            throw new IllegalArgumentException("Search requires at least three characters");
        }
        String needle = attendeeEmail.trim().toLowerCase(Locale.ROOT);
        return repository
            .findAllByEventId(eventId)
            .stream()
            .filter(ticket -> ticket.getAttendeeEmail().value().toLowerCase(Locale.ROOT).contains(needle))
            .limit(50)
            .map(ticket -> tickets.get(ticket.getId().value()))
            .toList();
    }

    public CheckInCounts counts(UUID eventId) {
        authorize(eventId);
        var eventTickets = repository.findAllByEventId(eventId);
        long registered = eventTickets
            .stream()
            .filter(ticket -> ticket.getStatus() == TicketStatus.VALID || ticket.getStatus() == TicketStatus.CHECKED_IN)
            .count();
        long checkedIn = eventTickets
            .stream()
            .filter(ticket -> ticket.getStatus() == TicketStatus.CHECKED_IN)
            .count();
        return new CheckInCounts(registered, checkedIn);
    }

    private UUID resolveTicketCode(UUID eventId, String input) {
        String code = input.strip();
        if (QR_CODE.matcher(code).matches()) return UUID.fromString(code);

        String manualCode = code.toUpperCase(Locale.ROOT).replaceAll("[\\s-]", "").replace('O', '0').replace('I', '1').replace('L', '1');
        if (!MANUAL_CODE.matcher(manualCode).matches()) {
            throw new IllegalArgumentException("Enter an eight-character check-in code or a ticket QR UUID");
        }

        return repository
            .findTicketCodeByEventIdAndManualCode(eventId, manualCode)
            .orElseThrow(() -> new TicketNotFoundException("Ticket not found for supplied ticket code"))
            .value();
    }

    private Long authorize(UUID eventId) {
        Objects.requireNonNull(eventId, "eventId cannot be null");
        Long userId = actor.requireUserId();
        if (!events.canCheckIn(eventId, userId)) {
            throw new AuthorizationException("User is not assigned to check in attendees for this event");
        }
        return userId;
    }

    public record CheckInCounts(long registered, long checkedIn) {}
}

@Service
@Transactional(readOnly = true)
public class TicketService {

    private final TicketRepository ticketRepository;
    private final EventInventoryRepository inventoryRepository;
    private final VerifiedUserEmailPort verifiedUserEmailPort;
    private final CurrentActor currentActor;
    private final EventManagementPort events;
    private final TransactionLocks locks;

    public TicketService(
        TicketRepository ticketRepository,
        EventInventoryRepository inventoryRepository,
        VerifiedUserEmailPort verifiedUserEmailPort,
        CurrentActor currentActor,
        EventManagementPort events,
        TransactionLocks locks
    ) {
        this.ticketRepository = Objects.requireNonNull(ticketRepository);
        this.inventoryRepository = Objects.requireNonNull(inventoryRepository);
        this.verifiedUserEmailPort = Objects.requireNonNull(verifiedUserEmailPort);
        this.currentActor = Objects.requireNonNull(currentActor);
        this.events = Objects.requireNonNull(events);
        this.locks = Objects.requireNonNull(locks);
    }

    @Transactional
    public void cancel(CancelTicketCommand command) {
        Objects.requireNonNull(command, "command cannot be null");

        Ticket ticket = requireTicketForUpdate(command.ticketId());

        boolean changed = ticket.cancel();
        if (!changed) {
            return;
        }

        EventInventory inventory = requireInventory(ticket.getEventId());

        if (ticket.getSeatId() == null) {
            inventory.reverseGASale(ticket.getTicketTypeId(), 1);
        } else {
            inventory.cancelSoldSeat(ticket.getSeatId(), ticket.getOrderId());
        }

        ticketRepository.save(ticket);
        inventoryRepository.save(inventory);
    }

    @Transactional
    public void refund(RefundTicketCommand command) {
        Objects.requireNonNull(command, "command cannot be null");

        Ticket ticket = requireTicketForUpdate(command.ticketId());

        boolean changed = ticket.markRefunded();
        if (!changed) {
            return;
        }

        EventInventory inventory = requireInventory(ticket.getEventId());

        if (ticket.getSeatId() == null) {
            inventory.reverseGASale(ticket.getTicketTypeId(), 1);
        } else {
            inventory.refundSoldSeat(ticket.getSeatId(), ticket.getOrderId());
        }

        ticketRepository.save(ticket);
        inventoryRepository.save(inventory);
    }

    @Transactional
    public TicketView checkIn(CheckInTicketCommand command) {
        Objects.requireNonNull(command, "command cannot be null");
        locks.lock("event-lifecycle:" + command.expectedEventId());
        requireAdmission(command.expectedEventId());

        Ticket ticket = requireTicketByCodeForUpdate(command.ticketCode());

        ensureExpectedEvent(ticket, command.expectedEventId());

        ticket.checkIn();

        return toView(ticketRepository.save(ticket));
    }

    @Transactional
    public TicketView reverseCheckIn(ReverseCheckInCommand command) {
        Objects.requireNonNull(command, "command cannot be null");
        locks.lock("event-lifecycle:" + command.expectedEventId());
        requireAdmission(command.expectedEventId());

        Ticket ticket = requireTicketByCodeForUpdate(command.ticketCode());

        ensureExpectedEvent(ticket, command.expectedEventId());

        ticket.reverseCheckIn();

        return toView(ticketRepository.save(ticket));
    }

    @Transactional
    public TicketView claim(ClaimTicketCommand command) {
        Objects.requireNonNull(command, "command cannot be null");

        Long userId = currentActor.requireUserId();
        Ticket ticket = requireTicketForUpdate(command.ticketId());

        String verifiedEmail = verifiedUserEmailPort
            .findVerifiedEmail(userId)
            .orElseThrow(() -> new AuthorizationException("Authenticated user does not have a verified email"));

        AttendeeEmail normalizedVerifiedEmail = new AttendeeEmail(verifiedEmail);

        if (!ticket.getAttendeeEmail().equals(normalizedVerifiedEmail)) {
            throw new AuthorizationException("Ticket attendee email does not match the authenticated user");
        }

        ticket.linkOwner(userId);

        return toView(ticketRepository.save(ticket));
    }

    public List<TicketView> getMine() {
        Long userId = currentActor.requireUserId();

        return ticketRepository.findAllByOwnerUserId(userId).stream().map(this::toView).toList();
    }

    public TicketView get(UUID ticketId) {
        return toView(requireTicket(ticketId));
    }

    /** Minimal ownership/context query for another bounded context's support workflow. */
    public Optional<TicketSupportReferenceView> findSupportReference(UUID ticketId) {
        Objects.requireNonNull(ticketId, "ticketId cannot be null");
        return ticketRepository.findById(new TicketId(ticketId)).map(ticket -> new TicketSupportReferenceView(

    public TicketValidationResult validate(UUID ticketCode, UUID expectedEventId) {
        Ticket ticket = requireTicketByCode(ticketCode);

        ensureExpectedEvent(ticket, expectedEventId);

        return new TicketValidationResult(
            ticket.getId().value(),
            ticket.getEventId(),
            ticket.getTicketTypeId().value(),
            ticket.getSeatId() == null ? null : ticket.getSeatId().value(),
            ticket.getStatus(),
            ticket.getStatus() == TicketStatus.VALID && allowsAdmission(ticket.getEventId())
        );
    }

    private void requireAdmission(UUID eventId) {
        var details = events.ticketDetails(eventId, null);
        if (details.cancelled()) throw new IllegalStateException("Event is cancelled");
        if (details.suspended()) throw new IllegalStateException("Event is suspended");
    }

    private boolean allowsAdmission(UUID eventId) {

        return !details.cancelled() && !details.suspended();
    }

    }

    }

        Objects.requireNonNull(rawTicketCode, "ticketCode cannot be null");

        TicketCode ticketCode = new TicketCode(rawTicketCode);

        return ticketRepository

            .orElseThrow(() -> new TicketNotFoundException("Ticket not found for supplied ticket code"));
    }

    private Ticket requireTicketByCodeForUpdate(UUID rawTicketCode) {
        Objects.requireNonNull(rawTicketCode, "ticketCode cannot be null");
        return ticketRepository
            .findByTicketCodeForUpdate(new TicketCode(rawTicketCode))
            .orElseThrow(() -> new TicketNotFoundException("Ticket not found for supplied ticket code"));
    }

        return inventoryRepository

    }

        if (!ticket.getEventId().equals(expectedEventId)) {
            throw new IllegalStateException("Ticket is not valid for the selected event");
        }
    }

            ticket.getOrderId().value(),

            ticket.getOwnerUserId(),
            ticket.getSeatId() == null ? null : ticket.getSeatId().value(),
            ticket.getTicketCode().value(),
            ticket.getManualCode(),
            ticket.getStatus(),
            ticket.getIssuedAt()
        );

    private float lines(
        PDPageContentStream content,
        PDType0Font font,
        String value,
        float size,
        float x,
        float y,
        float width,
        int maxLines
    ) throws IOException {
        var clean = new StringBuilder();
        for (int cp : value.codePoints().toArray()) {
            String glyph = Character.isISOControl(cp) || Character.isWhitespace(cp) ? " " : new String(Character.toChars(cp));
            try {
                font.encode(glyph);
                clean.append(glyph);
            } catch (IllegalArgumentException unsupported) {
                clean.append('?');
            }
        }
        String remaining = clean.toString().trim();
        for (int line = 0; line < maxLines && !remaining.isEmpty(); line++) {
            int end = 0;
            while (end < remaining.length()) {
                int next = remaining.offsetByCodePoints(end, 1);
                if ((font.getStringWidth(remaining.substring(0, next)) * size) / 1000 > width) break;
                end = next;
            }
            if (end == 0) end = remaining.offsetByCodePoints(0, 1);
            if (end < remaining.length()) {
                int space = remaining.lastIndexOf(' ', end);
                if (space > 0) end = space;
            }
            String text = remaining.substring(0, end).stripTrailing();
            remaining = remaining.substring(end).stripLeading();
            if (line == maxLines - 1 && !remaining.isEmpty()) {
                while (!text.isEmpty() && (font.getStringWidth(text + "...") * size) / 1000 > width)
                    text = text.substring(0, text.offsetByCodePoints(text.length(), -1));
                text += "...";
            }
            content.beginText();
            content.setFont(font, size);
            content.newLineAtOffset(x, y);
            content.showText(text);
            content.endText();
            y -= size + 4;
        }
        return y;
    }

import jakarta.persistence.PersistenceContext;
import java.util.List;
import java.util.Objects;
import java.util.Optional;
import java.util.UUID;
import org.springframework.stereotype.Repository;
import org.springframework.transaction.annotation.Transactional;

@Repository
@Transactional(readOnly = true)
public class JpaTicketRepository implements TicketRepository {

    private final SpringDataTicketJpaRepository jpaRepository;
    private final TicketPersistenceMapper mapper;

    @PersistenceContext
    private EntityManager entityManager;

    public JpaTicketRepository(SpringDataTicketJpaRepository jpaRepository, TicketPersistenceMapper mapper) {
        this.jpaRepository = Objects.requireNonNull(jpaRepository);
        this.mapper = Objects.requireNonNull(mapper);
    }

    @Override
    @Transactional
    public Ticket save(Ticket ticket) {
        var entity = saveEntity(ticket);
        if (entity.getManualCode() == null) entityManager.flush();
        return mapper.toDomain(entity);
    }

    @Override
    @Transactional
    public List<Ticket> saveAll(List<Ticket> tickets) {
        Objects.requireNonNull(tickets, "tickets cannot be null");
        var entities = tickets.stream().map(this::saveEntity).toList();
        if (entities.stream().anyMatch(entity -> entity.getManualCode() == null)) entityManager.flush();
        return entities.stream().map(mapper::toDomain).toList();
    }

    private TicketJpaEntity saveEntity(Ticket ticket) {
        Objects.requireNonNull(ticket, "ticket cannot be null");
        var managed = entityManager.find(TicketJpaEntity.class, ticket.getId().value());
        if (managed == null) {
            var created = mapper.toNewEntity(ticket);
            entityManager.persist(created);
            return created;
        }
        mapper.updateEntity(ticket, managed);
        return managed;
    }

    @Override
    public Optional<Ticket> findById(TicketId id) {
        Objects.requireNonNull(id, "TicketId cannot be null");
        return jpaRepository.findById(id.value()).map(mapper::toDomain);
    }

    @Override
    public Optional<Ticket> findByTicketCode(TicketCode ticketCode) {
        Objects.requireNonNull(ticketCode, "TicketCode cannot be null");
        return jpaRepository.findByTicketCode(ticketCode.value()).map(mapper::toDomain);
    }

    @Override
    public Optional<TicketCode> findTicketCodeByEventIdAndManualCode(UUID eventId, String manualCode) {
        return jpaRepository.findTicketCodeByEventIdAndManualCode(eventId, manualCode).map(TicketCode::new);
    }

        Objects.requireNonNull(ticketCode, "TicketCode cannot be null");
        return jpaRepository.findByTicketCodeForUpdate(ticketCode.value()).map(entity -> {

    static String decode(BufferedImage image) throws Exception {
        int width = image.getWidth(),
            height = image.getHeight();
        int[] pixels = image.getRGB(0, 0, width, height, null, 0, width);
        // A payload can contain extra finder-like patterns. Scan all candidates rather than
        // letting the single-code reader stop at an incorrect group of three patterns.
        var results = new QRCodeMultiReader().decodeMultiple(
            new BinaryBitmap(new HybridBinarizer(new RGBLuminanceSource(width, height, pixels))),
            Map.of(DecodeHintType.TRY_HARDER, true)
        );
        assertEquals(1, results.length, "Expected exactly one decodable admission QR code");
        return results[0].getText();
    }

            for (int dpi : List.of(96, 150, 300)) {

            }
        }
    }
