package com.biletflow.biletflow.ticketinventory.application.delivery;

import com.biletflow.biletflow.common.application.AuthorizationException;
import com.biletflow.biletflow.common.application.EntityNotFoundException;
import com.biletflow.biletflow.common.security.CurrentActor;
import com.biletflow.biletflow.ticketinventory.application.eventmanagement.port.EventManagementPort;
import com.biletflow.biletflow.ticketinventory.application.ticket.port.VerifiedUserEmailPort;
import com.biletflow.biletflow.ticketinventory.application.ticket.view.TicketView;
import com.biletflow.biletflow.ticketinventory.domain.ticket.*;
import com.biletflow.biletflow.ticketinventory.domain.tickettype.TicketTypeRepository;
import java.time.Clock;
import java.util.Objects;
import java.util.UUID;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
@Transactional(readOnly = true)
public class TicketDocumentService {

    private final TicketRepository tickets;
    private final TicketTypeRepository types;
    private final EventManagementPort events;
    private final VerifiedUserEmailPort emails;
    private final CurrentActor actor;
    private final TicketDocumentRenderer renderer;
    private final TicketDeliveryStore deliveries;
    private final Clock clock;

    public TicketDocumentService(
        TicketRepository tickets,
        TicketTypeRepository types,
        EventManagementPort events,
        VerifiedUserEmailPort emails,
        CurrentActor actor,
        TicketDocumentRenderer renderer,
        TicketDeliveryStore deliveries,
        Clock clock
    ) {
        this.tickets = Objects.requireNonNull(tickets);
        this.types = Objects.requireNonNull(types);
        this.events = Objects.requireNonNull(events);
        this.emails = Objects.requireNonNull(emails);
        this.actor = Objects.requireNonNull(actor);
        this.renderer = Objects.requireNonNull(renderer);
        this.deliveries = Objects.requireNonNull(deliveries);
        this.clock = Objects.requireNonNull(clock);
    }

    public byte[] pdf(UUID ticketId) {
        return renderer.pdf(document(authorize(ticketId)));
    }

    public byte[] qr(UUID ticketId) {
        return renderer.qr(document(authorize(ticketId)));
    }

    public String admissionCode(UUID ticketId) {
        var ticket = authorize(ticketId);
        String code = ticket.getManualCode();
        if (ticket.getStatus() != TicketStatus.VALID || code == null) {
            throw new IllegalStateException("This ticket has no usable check-in code");
        }
        var event = events.ticketDetails(ticket.getEventId(), null);
        if (event.cancelled() || event.suspended()) {
            throw new IllegalStateException("This ticket has no usable check-in code");
        }
        return code;
    }

    @Transactional
    public void requestEmail(UUID ticketId) {
        var ticket = authorize(ticketId);
        var document = document(ticket);
        if (ticket.getStatus() != TicketStatus.VALID || document.event().cancelled()) {
            throw new IllegalStateException("Only a valid ticket for a non-cancelled event can be emailed");
        }
        deliveries.request(ticketId, clock.instant());
    }

    /** Internal worker query; no public endpoint delegates directly to this method. */
    public TicketDocument forDelivery(UUID ticketId) {
        return document(requireTicket(ticketId));
    }

    private Ticket authorize(UUID ticketId) {
        Long user = actor.requireUserId();
        var ticket = requireTicket(ticketId);
        boolean recipient = emails
            .findVerifiedEmail(user)
            .map(email -> new AttendeeEmail(email).equals(ticket.getAttendeeEmail()))
            .orElse(false);
        if (!user.equals(ticket.getOwnerUserId()) && !recipient && !events.canManageTicketing(ticket.getEventId(), user)) {
            throw new AuthorizationException("Ticket is private");
        }
        return ticket;
    }

    private Ticket requireTicket(UUID id) {
        return tickets.findById(new TicketId(id)).orElseThrow(() -> new EntityNotFoundException("Ticket not found: " + id));
    }

    private TicketDocument document(Ticket ticket) {
        var type = types.findById(ticket.getTicketTypeId()).orElseThrow(() -> new EntityNotFoundException("Ticket type not found"));
        var view = new TicketView(
            ticket.getId().value(),
            ticket.getTicketTypeId().value(),
            ticket.getEventId(),
            ticket.getOrderId().value(),
            ticket.getAttendeeEmail().value(),
            ticket.getOwnerUserId(),
            ticket.getSeatId() == null ? null : ticket.getSeatId().value(),
            ticket.getTicketCode().value(),
            ticket.getManualCode(),
            ticket.getStatus(),
            ticket.getIssuedAt()
        );
        return new TicketDocument(view, type.getName(), events.ticketDetails(ticket.getEventId(), view.seatId()));
    }
}

package com.biletflow.biletflow.ticketinventory.rest;

import com.biletflow.biletflow.ticketinventory.application.delivery.TicketDocumentService;
import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.media.Content;
import io.swagger.v3.oas.annotations.media.Schema;
import io.swagger.v3.oas.annotations.responses.ApiResponse;
import io.swagger.v3.oas.annotations.responses.ApiResponses;
import io.swagger.v3.oas.annotations.tags.Tag;
import java.util.Objects;
import java.util.UUID;
import org.springframework.http.*;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/tickets")
@Tag(name = "Ticket delivery", description = "Private ticket documents and delivery to the ticket's recorded attendee email.")
@ApiResponses({
    @ApiResponse(
        responseCode = "401",
        description = "Authentication required",
        content = @Content(schema = @Schema(implementation = ProblemDetail.class))
    ),
    @ApiResponse(
        responseCode = "403",
        description = "Caller is not the ticket owner, verified recipient, or event organizer",
        content = @Content(schema = @Schema(implementation = ProblemDetail.class))
    ),
    @ApiResponse(
        responseCode = "404",
        description = "Ticket not found",
        content = @Content(schema = @Schema(implementation = ProblemDetail.class))
    ),
    @ApiResponse(
        responseCode = "409",
        description = "Operation is unavailable for this ticket or email resend cooldown is active",
        content = @Content(schema = @Schema(implementation = ProblemDetail.class))
    ),
})
public class TicketDocumentsResource {

    private final TicketDocumentService documents;

    public TicketDocumentsResource(TicketDocumentService documents) {
        this.documents = Objects.requireNonNull(documents);
    }

    @GetMapping(value = "/{ticketId}/pdf", produces = MediaType.APPLICATION_PDF_VALUE)
    @Operation(
        summary = "Download a printable ticket",
        description = "A4 PDF with event details, attendee email, seat, status, and the canonical admission QR. Cancelled/refunded/pending-refund tickets have no admission QR."
    )
    @ApiResponse(
        responseCode = "200",
        description = "Printable ticket",
        content = @Content(mediaType = MediaType.APPLICATION_PDF_VALUE, schema = @Schema(type = "string", format = "binary"))
    )
    public ResponseEntity<byte[]> pdf(@PathVariable UUID ticketId) {
        return file(documents.pdf(ticketId), MediaType.APPLICATION_PDF, "ticket-" + ticketId + ".pdf");
    }

    @GetMapping(value = "/{ticketId}/qr", produces = MediaType.IMAGE_PNG_VALUE)
    @Operation(
        summary = "Download the admission QR image",
        description = "PNG encoding the existing ticket code, accepted by the online scanner. Printed and digital copies do not create additional admissions."
    )
    @ApiResponse(
        responseCode = "200",
        description = "Ticket QR image",
        content = @Content(mediaType = MediaType.IMAGE_PNG_VALUE, schema = @Schema(type = "string", format = "binary"))
    )
    public ResponseEntity<byte[]> qr(@PathVariable UUID ticketId) {
        return file(documents.qr(ticketId), MediaType.IMAGE_PNG, "ticket-" + ticketId + ".png");
    }

    @GetMapping(value = "/{ticketId}/admission-code", produces = MediaType.APPLICATION_JSON_VALUE)
    @Operation(
        summary = "Get the short check-in code",
        description = "Returns the manual entry code for a valid ticket using the same access rules as the ticket QR image."
    )
    @ApiResponse(responseCode = "200", description = "Short check-in code")
    public ResponseEntity<AdmissionCode> admissionCode(@PathVariable UUID ticketId) {
        return ResponseEntity.ok()
            .cacheControl(CacheControl.noStore().cachePrivate())
            .body(new AdmissionCode(documents.admissionCode(ticketId)));
    }

    public record AdmissionCode(String manualCode) {}

    @PostMapping("/{ticketId}/delivery")
    @Operation(
        summary = "Request ticket email delivery",
        description = "Always sends to the recorded attendee email; no recipient override. Pending requests are idempotent. Completed deliveries can be resent after a one-minute cooldown. SMTP delivery is asynchronous and retried on failure."
    )
    @ApiResponse(responseCode = "202", description = "Delivery queued or already pending", content = @Content)
    public ResponseEntity<Void> email(@PathVariable UUID ticketId) {
        documents.requestEmail(ticketId);
        return ResponseEntity.accepted().cacheControl(CacheControl.noStore()).build();
    }

    private ResponseEntity<byte[]> file(byte[] content, MediaType type, String filename) {
        return ResponseEntity.ok()
            .contentType(type)
            .contentLength(content.length)
            .cacheControl(CacheControl.noStore().cachePrivate())
            .header(HttpHeaders.CONTENT_DISPOSITION, ContentDisposition.attachment().filename(filename).build().toString())
            .body(content);
    }
}

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

import com.biletflow.biletflow.common.application.AuthorizationException;
import com.biletflow.biletflow.ticketinventory.application.eventmanagement.port.*;
import com.biletflow.biletflow.ticketinventory.application.ticket.port.VerifiedUserEmailPort;

import com.biletflow.biletflow.ticketinventory.domain.ticket.*;
import com.biletflow.biletflow.ticketinventory.domain.tickettype.*;
import java.time.*;
import java.util.*;
import org.junit.jupiter.api.Test;

class TicketDocumentServiceTest {

    private final TicketRepository tickets = mock(TicketRepository.class);
    private final TicketTypeRepository types = mock(TicketTypeRepository.class);
    private final EventManagementPort events = mock(EventManagementPort.class);
    private final VerifiedUserEmailPort emails = mock(VerifiedUserEmailPort.class);
    private final TicketDocumentRenderer renderer = mock(TicketDocumentRenderer.class);
    private final TicketDeliveryStore deliveries = mock(TicketDeliveryStore.class);
    private final Instant now = Instant.parse("2026-10-19T10:00:00Z");
    private final Ticket ticket = new Ticket(
        TicketId.generate(),
        TicketTypeId.generate(),
        UUID.randomUUID(),
        new OrderId(UUID.randomUUID()),
        new AttendeeEmail("recipient@example.com"),
        null,
        null,
        TicketCode.generate(),
        "7K9M2Q8R",
        TicketStatus.VALID,
        now
    );
    private final TicketDocumentService service = new TicketDocumentService(
        tickets,
        types,
        events,
        emails,
        () -> Optional.of(1L),
        renderer,
        deliveries,
        Clock.fixed(now, ZoneOffset.UTC)
    );

    TicketDocumentServiceTest() {
        when(tickets.findById(ticket.getId())).thenReturn(Optional.of(ticket));
        var type = mock(TicketType.class);
        when(type.getName()).thenReturn("General admission");
        when(types.findById(ticket.getTicketTypeId())).thenReturn(Optional.of(type));
        when(events.ticketDetails(ticket.getEventId(), null)).thenReturn(
            new TicketEventDetails("Event", now, now.plusSeconds(3600), "Venue", "Address", "General admission", false)
        );
    }

    @Test
    void verifiedRecipientCanDownloadBeforeClaimingTicket() {
        when(emails.findVerifiedEmail(1L)).thenReturn(Optional.of("RECIPIENT@example.com"));
        when(renderer.pdf(any())).thenReturn(new byte[] { 1, 2 });
        assertArrayEquals(new byte[] { 1, 2 }, service.pdf(ticket.getId().value()));
        assertEquals("7K9M2Q8R", service.admissionCode(ticket.getId().value()));
        service.requestEmail(ticket.getId().value());
        verify(deliveries).request(ticket.getId().value(), now);
    }

    @Test
    void linkedOwnerCanRead() {
        ticket.linkOwner(1L);
        service.qr(ticket.getId().value());
        verify(renderer).qr(argThat(document -> document.ticket().ticketCode().equals(ticket.getTicketCode().value())));
    }

    @Test
    void organizerCanReadTicketsWithoutTakingOwnership() {
        when(events.canManageTicketing(ticket.getEventId(), 1L)).thenReturn(true);
        service.pdf(ticket.getId().value());
        verify(renderer).pdf(any());
    }

    @Test
    void anotherUserCannotDownloadOrRequestEmail() {
        when(emails.findVerifiedEmail(1L)).thenReturn(Optional.of("someone-else@example.com"));
        assertThrows(AuthorizationException.class, () -> service.pdf(ticket.getId().value()));
        assertThrows(AuthorizationException.class, () -> service.qr(ticket.getId().value()));
        assertThrows(AuthorizationException.class, () -> service.admissionCode(ticket.getId().value()));
        assertThrows(AuthorizationException.class, () -> service.requestEmail(ticket.getId().value()));
        verifyNoInteractions(renderer, deliveries);
        verify(events, never()).ticketDetails(any(), any());
    }

    @Test
    void refundPendingOrCancelledEventCannotTriggerDelivery() {
        ticket.linkOwner(1L);
        ticket.prepareRefund();
        assertThrows(IllegalStateException.class, () -> service.admissionCode(ticket.getId().value()));
        assertThrows(IllegalStateException.class, () -> service.requestEmail(ticket.getId().value()));
        verifyNoInteractions(deliveries);
    }

    @Test
    void cancelledEventCannotTriggerDelivery() {
        ticket.linkOwner(1L);
        when(events.ticketDetails(ticket.getEventId(), null)).thenReturn(
            new TicketEventDetails("Event", now, now.plusSeconds(3600), "Venue", "Address", "General admission", true)
        );
        assertThrows(IllegalStateException.class, () -> service.requestEmail(ticket.getId().value()));
        assertThrows(IllegalStateException.class, () -> service.admissionCode(ticket.getId().value()));
        verifyNoInteractions(deliveries);
    }
}

package com.biletflow.biletflow.ticketinventory.rest;

import static org.mockito.Mockito.*;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.*;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.*;

import com.biletflow.biletflow.common.application.AuthorizationException;
import com.biletflow.biletflow.ticketinventory.application.delivery.TicketDeliveryCooldownException;
import com.biletflow.biletflow.ticketinventory.application.delivery.TicketDocumentService;
import java.util.UUID;
import org.junit.jupiter.api.Test;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.setup.MockMvcBuilders;

class TicketDocumentsResourceTest {

    @Test
    void pdfUsesPrivateNonCacheableAttachmentResponse() throws Exception {
        var documents = mock(TicketDocumentService.class);
        var id = UUID.randomUUID();
        when(documents.pdf(id)).thenReturn(new byte[] { 1, 2, 3 });
        var mvc = MockMvcBuilders.standaloneSetup(new TicketDocumentsResource(documents)).build();
        mvc.perform(get("/api/tickets/{id}/pdf", id).accept(MediaType.APPLICATION_PDF))
            .andExpect(status().isOk())
            .andExpect(content().contentType(MediaType.APPLICATION_PDF))
            .andExpect(header().string("Content-Disposition", "attachment; filename=\"ticket-" + id + ".pdf\""))
            .andExpect(header().string("Cache-Control", "no-store, private"))
            .andExpect(content().bytes(new byte[] { 1, 2, 3 }));
    }

    @Test
    void pdfAcceptHeaderDoesNotHideAnAuthorizationFailure() throws Exception {
        var documents = mock(TicketDocumentService.class);
        var id = UUID.randomUUID();
        when(documents.pdf(id)).thenThrow(new AuthorizationException("Ticket is private"));
        var mvc = MockMvcBuilders.standaloneSetup(new TicketDocumentsResource(documents))
            .setControllerAdvice(new TicketInventoryExceptionHandler())
            .build();
        mvc.perform(get("/api/tickets/{id}/pdf", id).accept(MediaType.APPLICATION_PDF))
            .andExpect(status().isForbidden())
            .andExpect(content().contentTypeCompatibleWith(MediaType.APPLICATION_PROBLEM_JSON));
    }

    @Test
    void admissionCodeUsesPrivateNonCacheableResponse() throws Exception {
        var documents = mock(TicketDocumentService.class);
        var id = UUID.randomUUID();
        when(documents.admissionCode(id)).thenReturn("7K9M2Q8R");
        var mvc = MockMvcBuilders.standaloneSetup(new TicketDocumentsResource(documents)).build();
        mvc.perform(get("/api/tickets/{id}/admission-code", id))
            .andExpect(status().isOk())
            .andExpect(header().string("Cache-Control", "no-store, private"))
            .andExpect(jsonPath("$.manualCode").value("7K9M2Q8R"));
    }

    @Test
    void resendCooldownReturnsAConflict() throws Exception {
        var documents = mock(TicketDocumentService.class);
        var id = UUID.randomUUID();
        doThrow(new TicketDeliveryCooldownException()).when(documents).requestEmail(id);
        var mvc = MockMvcBuilders.standaloneSetup(new TicketDocumentsResource(documents))
            .setControllerAdvice(new TicketInventoryExceptionHandler())
            .build();
        mvc.perform(post("/api/tickets/{id}/delivery", id))
            .andExpect(status().isConflict())
            .andExpect(content().contentTypeCompatibleWith(MediaType.APPLICATION_PROBLEM_JSON))
            .andExpect(jsonPath("$.detail").value("Wait at least one minute before requesting another email"));
    }
}
